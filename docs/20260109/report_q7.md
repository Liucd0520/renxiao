### 5. 附录 - 使用的 SQL 查询
```sql
-- 查询指定客户名下的所有设备ID
SELECT ci.CI_ID
FROM t_bz_config_ci_ne_root ci
JOIN t_bz_config_customer c ON ci.CUSTOMER_ID = c.CUSTOMER_ID
WHERE c.CUSTOMER_NAME = '指定客户名' AND ci.IS_DELETED = 0;

-- 查询这些设备近一周内的告警类型、告警时间和恢复时间
SELECT 
    e.EVENT_NAME AS alarm_type,
    e.EVENT_TIME AS alarm_time,
    e.RECOVERY_TIME AS recovery_time
FROM 
    event_history e
WHERE 
    e.NE_ID IN (1001, 1002, 1003, 1004)
    AND e.EVENT_TIME >= DATE_SUB(NOW(), INTERVAL 1 WEEK)
    AND e.EVENT_STATUS_NAME = 'alarm';
```
