
CREATE TABLE `WH_MEMBER_T` (
	`TEMP_CODE` VARCHAR(100) CHARACTER  COMMENT '工时项目编号',  统计时经常需要COUNT( DISTINCT `TEMP_CODE`) 计算数量 
	`PROJECT_NAME` VARCHAR(200) CHARACTER  COMMENT '项目名称', 
	`MEMBER_ID` VARCHAR(255) CHARACTER  COMMENT '人员id', 
	`MEMBER_IN_TIME` DATE COMMENT '人员进入项目时间', 
	`NAME` VARCHAR(100) CHARACTER  COMMENT '员工姓名', 
	`DEPT` VARCHAR(64) CHARACTER  COMMENT '员工部门', 
	`DEPARTMENT_NAME` VARCHAR(100) CHARACTER  COMMENT '员工二级部门', 
	`STAFF_POST` VARCHAR(100) CHARACTER  COMMENT '员工岗位', 
	`PROJECTMEMBERWORKHOURS` BIGINT COMMENT '人员在该项目上所有时间填报的工时', 
	`MEMBERHOURS` BIGINT COMMENT '人员在所有项目上所有时间填报的总工时', 
	`WHMONTH` VARCHAR(6) CHARACTER  COMMENT '工时填报的月份',  由于是字符串格式，禁止对该列使用YEAR()，MONTH()时间函数
	`MONTHHOURS` BIGINT COMMENT '人员在该项目上当月填报的工时，0意味着该人员没有在该项目中填写工时', 
	`MEMBERMONTHHOURS` BIGINT COMMENT '人员在所有项目上当月填报的工时，0意味着该人员在任何项目中均没有填写工时'
)ENGINE=InnoDB COMMENT='以每名人员为主体，基于人员相关的信息' DEFAULT CHARSET=utf8mb4 COLLATE utf8mb4_0900_ai_ci

CREATE TABLE `WH_PROJECT_WH` (
	`TEMP_CODE` VARCHAR(100) CHARACTER  COMMENT '项目编号， 统计时经常需要COUNT( DISTINCT `TEMP_CODE`) 计算数量 
	`PROJECT_NAME` VARCHAR(200) CHARACTER  COMMENT '项目名称', 
	`PROJECT_TYPE` VARCHAR(6) CHARACTER  COMMENT '项目类型，包括：前评审项目, 研发类业务, 预投入业务,后端管理业务,开账类业务', 不是前评审项目的均属于自建项目
	`QPSYJ` VARCHAR(100) CHARACTER  COMMENT '前评审编号', 
	`MANAGER_ID` VARCHAR(50) CHARACTER  COMMENT '项目经理id', 
	`MANAGERNAME` VARCHAR(100) CHARACTER  COMMENT '项目经理名称', 
	`MAXWORKHOURS` VARCHAR(100) CHARACTER  COMMENT '项目的最大工时数', 
	`START_TIME` VARCHAR(10) CHARACTER  COMMENT '项目开始时间，格式为''YYYY-MM-DD''', 
	`END_TIME` VARCHAR(10) CHARACTER  COMMENT '项目结束时间，格式为''YYYY-MM-DD''', 
	`DEPT` VARCHAR(64) CHARACTER  COMMENT '部门，包括：大数据事业部，研发中心，运营商数智事业部，智能云网事业部，智慧城市事业部，金融数智事业部，智能视频事业部，工业数智事业部，智能通信事业部，信创与安全事业部，市场部，人力资源部', 
	`DEPARTMENT_NAME` VARCHAR(100) CHARACTER  COMMENT '二级部门', 
	`WHMONTH` VARCHAR(6) CHARACTER  COMMENT '工时所在月份，格式为''YYYYMM''', 
	`PROJECTWORKHOURS` BIGINT COMMENT '所有人员填写的当月项目工时数', 
	`SRS` BIGINT COMMENT '收入', 
	`CBS` BIGINT COMMENT '成本', 
	`CCB` BIGINT COMMENT '投入产出比：（收入-成本）/工时数'
)ENGINE=InnoDB COMMENT='以项目为统计基准，基于每个项目在每个月的工时填报信息，与人员无关' DEFAULT CHARSET=utf8mb4 COLLATE utf8mb4_0900_ai_ci
